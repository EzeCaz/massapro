   117→# ───────────────────── Helpers ─────────────────────
   118→def add_heading(text, style, level=0):
   119→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
   120→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
   121→    p.bookmark_name = text
   122→    p.bookmark_level = level
   123→    p.bookmark_text = text
   124→    p.bookmark_key = key
   125→    return p

   127→MAX_KEEP_H = A4[1] * 0.4
   128→def safe_keep(elements):
   129→    total = 0
   130→    for el in elements:
   131→        try:
   132→            w, h = el.wrap(AVAIL_W, A4[1])
   133→            total += h
   134→        except Exception:
   135→            total += 50
   136→    if total <= MAX_KEEP_H:
   137→        return [KeepTogether(elements)]
   138→    elif len(elements) >= 2:
   139→        return [KeepTogether(elements[:2])] + list(elements[2:])
   140→    else:
   141→        return list(elements)

   166→def embed_image(path, max_width=None, max_height=None):
   167→    if max_width is None:
   168→        max_width = AVAIL_W
   169→    if max_height is None:
   170→        max_height = A4[1] * 0.55
   171→    img = PILImage.open(path)
   172→    w, h = img.size
   173→    rw = max_width / w if w > max_width else 1.0
   174→    rh = max_height / h if h > max_height else 1.0
   175→    r = min(rw, rh)
   176→    return Image(path, width=w * r, height=h * r)

   178→def callout(stat, label, width=140):
   179→    t = Table(
   180→        [[Paragraph('<b>%s</b>' % stat, stat_big)],
   181→         [Paragraph(label, stat_lbl)]],
   182→        colWidths=[width]

   184→    t.setStyle(TableStyle([
   185→        ('BACKGROUND', (0,0), (-1,-1), ACCENT_ULTRA),
   186→        ('BOX', (0,0), (-1,-1), 1, ACCENT),
   187→        ('LINEABOVE', (0,1), (-1,1), 0.5, BORDER_LIGHT),
   188→        ('TOPPADDING', (0,0), (-1,-1), 10),
   189→        ('BOTTOMPADDING', (0,0), (-1,-1), 10),
   190→        ('LEFTPADDING', (0,0), (-1,-1), 6),
   191→        ('RIGHTPADDING', (0,0), (-1,-1), 6),
   192→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   193→    ]))
   194→    return t

   196→def callout_row(items):
   197→    """Row of stat callouts, evenly spaced."""
   198→    n = len(items)
   199→    each_w = AVAIL_W / n
   200→    cells = []
   201→    for stat, lbl in items:
   202→        c = callout(stat, lbl, width=each_w - 10)
   203→        cells.append(c)
   204→    outer = Table([cells], colWidths=[each_w] * n, hAlign='CENTER')
   205→    outer.setStyle(TableStyle([
   206→        ('VALIGN', (0,0), (-1,-1), 'MIDDLE'),
   207→        ('LEFTPADDING', (0,0), (-1,-1), 4),
   208→        ('RIGHTPADDING', (0,0), (-1,-1), 4),
   209→        ('TOPPADDING', (0,0), (-1,-1), 0),
   210→        ('BOTTOMPADDING', (0,0), (-1,-1), 0),
   211→    ]))
   212→    return outer

   214→def std_table(data_rows, col_ratios, header=True):
   215→    """data_rows: list of lists of strings (already-plain). First row is header if header=True.
   216→    col_ratios: list of floats summing to 1.0"""
   217→    col_widths = [r * AVAIL_W for r in col_ratios]
   218→    # Wrap all cells in Paragraph
   219→    rows = []
   220→    for ri, row in enumerate(data_rows):
   221→        prow = []
   222→        for ci, cell in enumerate(row):
   223→            if header and ri == 0:
   224→                prow.append(Paragraph('<b>%s</b>' % cell, th_style))
   225→            else:
   226→                prow.append(Paragraph(cell, td_style))
   227→        rows.append(prow)
   228→    t = Table(rows, colWidths=col_widths, hAlign='CENTER', repeatRows=1 if header else 0)
   229→    style = [
   230→        ('VALIGN', (0,0), (-1,-1), 'TOP'),
   231→        ('LEFTPADDING', (0,0), (-1,-1), 7),
   232→        ('RIGHTPADDING', (0,0), (-1,-1), 7),
   233→        ('TOPPADDING', (0,0), (-1,-1), 6),
   234→        ('BOTTOMPADDING', (0,0), (-1,-1), 6),
   235→        ('GRID', (0,0), (-1,-1), 0.4, BORDER_LIGHT),
   236→    ]
   237→    if header:
   238→        style += [
   239→            ('BACKGROUND', (0,0), (-1,0), TABLE_HEADER_COLOR),
   240→            ('TEXTCOLOR', (0,0), (-1,0), colors.white),
   241→            ('TOPPADDING', (0,0), (-1,0), 8),
   242→            ('BOTTOMPADDING', (0,0), (-1,0), 8),
   243→        ]
   244→        # Alternating rows
   245→        for i in range(1, len(rows)):
   246→            bg = TABLE_ROW_ODD if (i % 2 == 1) else TABLE_ROW_EVEN
   247→            style.append(('BACKGROUND', (0,i), (-1,i), bg))
   248→    t.setStyle(TableStyle(style))
   249→    return t
