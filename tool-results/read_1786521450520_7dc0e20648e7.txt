    56→// ───────────────────── Title layout helpers ─────────────────────
    57→function splitTitleLines(title, charsPerLine) {
    58→  if (title.length <= charsPerLine) return [title];
    59→  const breakAfter = new Set([...' -_—–·/\t,.;:!?']);
    60→  const lines = [];
    61→  let remaining = title;
    62→  while (remaining.length > charsPerLine) {
    63→    let breakAt = -1;
    64→    for (let i = charsPerLine; i >= Math.floor(charsPerLine * 0.6); i--) {
    65→      if (i < remaining.length && breakAfter.has(remaining[i - 1])) {
    66→        breakAt = i; break;
    67→      }
    68→    }
    69→    if (breakAt === -1) {
    70→      const limit = Math.min(remaining.length, Math.ceil(charsPerLine * 1.3));
    71→      for (let i = charsPerLine + 1; i < limit; i++) {
    72→        if (breakAfter.has(remaining[i - 1])) { breakAt = i; break; }
    73→      }
    74→    }
    75→    if (breakAt === -1) breakAt = charsPerLine;
    76→    lines.push(remaining.slice(0, breakAt).trim());
    77→    remaining = remaining.slice(breakAt).trim();
    78→  }
    79→  if (remaining) lines.push(remaining);
    80→  return lines;
    81→}

    83→function calcTitleLayout(title, maxWidthTwips, preferredPt = 40, minPt = 24) {
    84→  const charsPerLine = (pt) => Math.floor(maxWidthTwips / (pt * 20));
    85→  let titlePt = preferredPt;
    86→  let lines;
    87→  while (titlePt >= minPt) {
    88→    const cpl = charsPerLine(titlePt);
    89→    if (cpl < 2) { titlePt -= 2; continue; }
    90→    lines = splitTitleLines(title, cpl);
    91→    if (lines.length <= 3) break;
    92→    titlePt -= 2;
    93→  }
    94→  if (!lines || lines.length > 3) {
    95→    lines = splitTitleLines(title, charsPerLine(minPt));
    96→    titlePt = minPt;
    97→  }
    98→  return { titlePt, titleLines: lines };
    99→}

   101→function calcCoverSpacing(params) {
   102→  const { titleLineCount = 1, titlePt = 36, hasSubtitle = false,
   103→          hasEnglishLabel = false, metaLineCount = 0,
   104→          fixedHeight = 800, pageHeight = 16838,
   105→          marginTop = 0, marginBottom = 0 } = params;
   106→  const SAFETY = 1200;
   107→  const usableHeight = pageHeight - marginTop - marginBottom - SAFETY;
   108→  const titleHeight = titleLineCount * (titlePt * 23 + 200);
   109→  const subtitleHeight = hasSubtitle ? (12 * 23 + 600) : 0;
   110→  const englishLabelHeight = hasEnglishLabel ? (9 * 23 + 600) : 0;
   111→  const metaHeight = metaLineCount * (10 * 23 + 100);
   112→  const implicitParaHeight = 3 * 300;
   113→  const contentHeight = titleHeight + subtitleHeight + englishLabelHeight +
   114→                        metaHeight + fixedHeight + implicitParaHeight;
   115→  const remainingSpace = usableHeight - contentHeight;
   116→  const safeRemaining = Math.max(remainingSpace, 400);
   117→  const FOOTER_MIN = 800;
   118→  const rawTop = Math.floor(safeRemaining * 0.45);
   119→  const rawBottom = Math.floor(safeRemaining * 0.45);
   120→  const bottomSpacing = Math.max(rawBottom, FOOTER_MIN);
   121→  const topSpacing = Math.max(rawTop - Math.max(0, FOOTER_MIN - rawBottom), 400);
   122→  return { topSpacing, bottomSpacing };
   123→}

   327→// ───────────────────── Slide builder ─────────────────────
   328→// Each slide becomes a structured section in the document.
   329→function slide(num, title, keyMessage, bodyContent, speakerNotes, visualRecs) {
   330→  const out = [];
   331→  out.push(h2(`Slide ${num}: ${title}`));
   332→
   333→  out.push(h3("Key Message (on-slide headline)"));
   334→  out.push(calloutBox(keyMessage));
   335→  out.push(spacer(120));
   336→
   337→  out.push(h3("Body Content (on-slide bullets)"));
   338→  for (const item of bodyContent) {
   339→    if (typeof item === "string") {
   340→      out.push(bullet(item));
   341→    } else if (Array.isArray(item)) {
   342→      if (item.length > 0 && typeof item[0] === "string") {
   343→        // [main, [sub-bullets]] pattern
   344→        out.push(bullet(item[0]));
   345→        if (Array.isArray(item[1])) {
   346→          for (const sub of item[1]) out.push(bullet(sub, 1));
   347→        }
   348→      } else {
   349→        // Array of inline rich segments [{text, bold?}, {text, italics?}]
   350→        out.push(bulletRich(item));
   351→      }
   352→    } else if (item.rich) {
   353→      out.push(bulletRich(item.rich));
   354→    }
   355→  }
   356→  out.push(spacer(120));
   357→
   358→  out.push(h3("Speaker Notes"));
   359→  for (const para of speakerNotes) {
   360→    out.push(body(para));
   361→  }
   362→  out.push(spacer(120));
   363→
   364→  out.push(h3("Visual Recommendations"));
   365→  for (const item of visualRecs) {
   366→    out.push(bullet(item));
   367→  }
   368→  out.push(spacer(240));
   369→
   370→  return out;
   371→}
